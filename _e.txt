            concluido: '🏁 Concluído'
        };

        const PESO_STATUS_BOOK = { pendente: 0, confirmado: 1, concluido: 2, recusado: 3, cancelado: 4 };

        function cardSolicitacao(b, souTerapeuta) {
            const rot = ROTULO_STATUS_BOOK[b.status] || esc(b.status);
            const quando = rotuloData(b.data).texto + ' às ' + b.hora_inicio;

            const quem = souTerapeuta
                ? esc(b.consulente_nome || 'Consulente') +
                  (b.consulente_contato ? ' · ' + esc(b.consulente_contato) : '')
                : esc(b.terapeuta_nome || 'Terapeuta');

            const servico = b.servico_nome
                ? esc(b.servico_nome) + (b.servico_preco ? ' · ' + esc(b.servico_preco) : '')
                : 'Atendimento a combinar';

            let prazo = '';
            if (b.status === 'pendente' && b.minutos_para_expirar != null) {
                prazo = b.minutos_para_expirar >= 60
                    ? 'expira em ' + Math.round(b.minutos_para_expirar / 60) + 'h'
                    : 'expira em ' + b.minutos_para_expirar + 'min';
            }

            let acoes = '';
            if (souTerapeuta && b.status === 'pendente') {
                acoes = '<button type="button" class="tp-btn-sec" data-book="confirmar" data-id="' + esc(b.id) + '">Confirmar</button>' +
                        '<button type="button" class="tp-btn-sec" data-book="recusar" data-id="' + esc(b.id) + '">Recusar</button>';
            } else if (souTerapeuta && b.status === 'confirmado') {
                acoes = '<button type="button" class="tp-btn-sec" data-book="concluir" data-id="' + esc(b.id) + '">Marcar como realizado</button>';
            } else if (!souTerapeuta && (b.status === 'pendente' || b.status === 'confirmado')) {
                acoes = '<button type="button" class="tp-btn-sec" data-book="cancelar" data-id="' + esc(b.id) + '">Cancelar</button>';
            }

            return `
                <div class="tp-book">
                    <div class="tp-book-topo">
                        <span class="tp-book-quando">${esc(quando)}</span>
                        <span class="tp-book-status s-${esc(b.status)}">${rot}</span>
                    </div>
                    <div class="tp-book-info">${quem}<br>${servico} · ${esc(b.duracao_min)} min</div>
                    ${prazo ? '<div class="tp-book-prazo">⏳ ' + prazo + '</div>' : ''}
                    ${b.observacao ? '<div class="tp-book-obs">“' + esc(b.observacao) + '”</div>' : ''}
                    ${b.resposta ? '<div class="tp-book-resp">Resposta: ' + esc(b.resposta) + '</div>' : ''}
                    ${acoes ? '<div class="tp-book-acoes">' + acoes + '</div>' : ''}
                </div>
            `;
        }
